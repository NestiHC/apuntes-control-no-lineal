\section*{Temario}
\addcontentsline{toc}{section}{Temario}

\begin{enumerate}[label=\arabic*.,leftmargin=*]
    \item \textbf{Introducción}
    \begin{enumerate}[label*=\arabic*.,leftmargin=*]
        \item Control no lineal y robusto
        \item LMI y complemento de Schur
    \end{enumerate}

    \item \textbf{Estabilidad}
    \begin{enumerate}[label*=\arabic*.,leftmargin=*]
        \item Sistemas autónomos
        \item El principio de invariancia
        \item Teoremas de estabilidad
        \begin{enumerate}[label*=\arabic*.,leftmargin=*]
            \item Primer teorema de estabilidad de Lyapunov
            \item Segundo teorema de estabilidad de Lyapunov
        \end{enumerate}
        \item Estabilidad asintótica, exponencial, global y uniformidad
    \end{enumerate}

    \item \textbf{Linealización}
    \begin{enumerate}[label*=\arabic*.,leftmargin=*]
        \item Entrada--salida
        \item De estado completo
        \item Feedback linearization
        \item Derivadas de Lie
        \item Grado relativo
        \item Desacoplamiento entrada--salida
    \end{enumerate}

    \item \textbf{Control robusto}
    \begin{enumerate}[label*=\arabic*.,leftmargin=*]
        \item Control por modos deslizantes
        \begin{enumerate}[label*=\arabic*.,leftmargin=*]
            \item Superficie de deslizamiento
            \item Dinámica de llegada
            \item Condición de deslizamiento
            \item Ley de alcance
            \item Condiciones de suficiencia
            \item Chattering
        \end{enumerate}
        \item Atenuación de chattering
        \begin{enumerate}[label*=\arabic*.,leftmargin=*]
            \item Pseudomodo deslizante
            \item SMC asintótico
            \item Super-twisting
            \item Atenuación mediante observadores (ADRC)
        \end{enumerate}
    \end{enumerate}

    \item \textbf{Observadores de estado}
    \begin{enumerate}[label*=\arabic*.,leftmargin=*]
        \item Observadores de Luenberger
        \item Observadores de estado extendido (LESO)
        \item Observador por modos deslizantes
        \item Observadores de perturbaciones
    \end{enumerate}

    \item \textbf{PID general}
    \begin{enumerate}[label*=\arabic*.,leftmargin=*]
        \item Derivadas filtradas
        \item GPID para sistemas lineales
        \item GPID para sistemas no lineales
    \end{enumerate}

    \item \textbf{Sincronización de sistemas}
    \begin{enumerate}[label*=\arabic*.,leftmargin=*]
        \item Sincronización completa
        \item Active--passive decomposition
        \item Elemento primitivo
        \item Sincronización generalizada
    \end{enumerate}
\end{enumerate}
